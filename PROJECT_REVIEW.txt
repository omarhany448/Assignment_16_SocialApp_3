import { createCipheriv, createDecipheriv, randomBytes } from 'node:crypto'
import { env } from '../../config/env.js'

const algorithm = 'aes-256-gcm'

const getEncryptionKey = (): Buffer => {
    const key = Buffer.from(env.ENCRYPTION_KEY, 'hex')

    if (key.length !== 32) {
        throw new Error('ENCRYPTION_KEY must be a 64-character hexadecimal value')
    }

    return key
}

export const encrypt = (value: string): string => {
    const iv = randomBytes(12)
    const cipher = createCipheriv(algorithm, getEncryptionKey(), iv)

    const encrypted = Buffer.concat([
        cipher.update(value, 'utf8'),
        cipher.final()
    ])

    const authTag = cipher.getAuthTag()

    return [
        iv.toString('hex'),
        authTag.toString('hex'),
        encrypted.toString('hex')
    ].join(':')
}

export const decrypt = (encryptedValue: string): string => {
    const [ivValue, authTagValue, encryptedText] = encryptedValue.split(':')

    if (!ivValue || !authTagValue || !encryptedText) {
        throw new Error('Invalid encrypted value')
    }

    const decipher = createDecipheriv(
        algorithm,
        getEncryptionKey(),
        Buffer.from(ivValue, 'hex')
    )

    decipher.setAuthTag(Buffer.from(authTagValue, 'hex'))

    const decrypted = Buffer.concat([
        decipher.update(Buffer.from(encryptedText, 'hex')),
        decipher.final()
    ])

    return decrypted.toString('utf8')
}

import { createHmac, timingSafeEqual } from 'node:crypto'
import { env } from '../../config/env.js'
import { UnauthorizedException } from '../exceptions/error.exception.js'

interface TokenPayload {
    sub: string
    exp: number
}

const getSecret = (): string => {
    if (!env.ACCESS_TOKEN_SECRET || env.ACCESS_TOKEN_SECRET.length < 32) {
        throw new Error('ACCESS_TOKEN_SECRET must contain at least 32 characters')
    }

    return env.ACCESS_TOKEN_SECRET
}

export const generateAccessToken = (userId: string): string => {
    const header = Buffer.from(
        JSON.stringify({ alg: 'HS256', typ: 'JWT' })
    ).toString('base64url')

    const payload: TokenPayload = {
        sub: userId,
        exp: Math.floor(Date.now() / 1000) + 3600
    }

    const encodedPayload = Buffer.from(
        JSON.stringify(payload)
    ).toString('base64url')

    const data = `${header}.${encodedPayload}`

    const signature = createHmac('sha256', getSecret())
        .update(data)
        .digest('base64url')

    return `${data}.${signature}`
}

export const verifyAccessToken = (token: string): TokenPayload => {
    try {
        const parts = token.split('.')

        if (parts.length !== 3) {
            throw new Error()
        }

        const [header, payload, signature] = parts

        if (!header || !payload || !signature) {
            throw new Error()
        }

        const decodedHeader = JSON.parse(
            Buffer.from(header, 'base64url').toString('utf8')
        )

        if (decodedHeader.alg !== 'HS256' || decodedHeader.typ !== 'JWT') {
            throw new Error()
        }

        const data = `${header}.${payload}`

        const expectedSignature = createHmac('sha256', getSecret())
            .update(data)
            .digest()

        const actualSignature = Buffer.from(signature, 'base64url')

        if (
            expectedSignature.length !== actualSignature.length ||
            !timingSafeEqual(expectedSignature, actualSignature)
        ) {
            throw new Error()
        }

        const decodedPayload = JSON.parse(
            Buffer.from(payload, 'base64url').toString('utf8')
        ) as TokenPayload

        if (
            typeof decodedPayload.sub !== 'string' ||
            typeof decodedPayload.exp !== 'number' ||
            decodedPayload.exp <= Math.floor(Date.now() / 1000)
        ) {
            throw new Error()
        }

        return decodedPayload
    } catch {
        throw new UnauthorizedException('Invalid or expired access token')
    }
}

import type { RequestHandler } from 'express'
import { UnauthorizedException } from '../common/exceptions/error.exception.js'
import { verifyAccessToken } from '../common/security/token.security.js'

export const authenticationMiddleware: RequestHandler = (
    req,
    res,
    next
) => {
    try {
        const authorization = req.headers.authorization

        if (!authorization?.startsWith('Bearer ')) {
            throw new UnauthorizedException('Access token is required')
        }

        const token = authorization.slice(7)
        const payload = verifyAccessToken(token)

        req.user = {
            id: payload.sub
        }

        next()
    } catch (error) {
        next(error)
    }
}

import type { RequestHandler } from 'express'
import { z } from 'zod'
import { BadRequestException } from '../common/exceptions/error.exception.js'

export const validationMiddleware = (
    schema: z.ZodType
): RequestHandler => {
    return (req, res, next) => {
        const result = schema.safeParse({
            body: req.body,
            params: req.params,
            query: req.query
        })

        if (!result.success) {
            return next(
                new BadRequestException(
                    'Validation Error',
                    result.error.issues
                )
            )
        }

        next()
    }
}

import { randomInt } from 'node:crypto'
import { BaseRepository } from '../../common/repository/index.js'
import { hashPassword, comparePassword } from '../../common/security/index.js'
import { generateAccessToken } from '../../common/security/token.security.js'
import { sendEmail } from '../../common/utils/email.utils.js'
import { redis } from '../../common/redis/redis.client.js'
import {
    ConflictException,
    BadRequestException,
    NotFoundException,
    UnauthorizedException
} from '../../common/exceptions/error.exception.js'
import { UserModel } from '../../DB/model/user.model.js'
import type {
    SignupDTO,
    ConfirmEmailDTO,
    LoginDTO
} from './authentication.dto.js'

const userRepository = new BaseRepository(UserModel)

export const signup = async (data: SignupDTO) => {
    const email = data.email.toLowerCase()

    const existingUser = await userRepository.findOne({ email })

    if (existingUser) {
        throw new ConflictException('Email already exists')
    }

    const hashedPassword = await hashPassword(data.password)

    const user = await userRepository.create({
        firstName: data.firstName,
        lastName: data.lastName,
        email,
        password: hashedPassword,
        isVerified: false
    })

    const code = randomInt(100000, 1000000).toString()

    await redis.set(`email-verification:${email}`, code, 'EX', 300)

    await sendEmail(
        email,
        'Email Verification',
        `Your verification code is ${code}. It expires in 5 minutes.`
    )

    return {
        id: user._id.toString(),
        firstName: user.firstName,
        lastName: user.lastName,
        email: user.email,
        isVerified: user.isVerified
    }
}

export const confirmEmail = async (data: ConfirmEmailDTO) => {
    const email = data.email.toLowerCase()

    const user = await userRepository.findOne({ email })

    if (!user) {
        throw new NotFoundException('User not found')
    }

    if (user.isVerified) {
        throw new BadRequestException('Email is already verified')
    }

    const savedCode = await redis.get(`email-verification:${email}`)

    if (!savedCode || savedCode !== data.code) {
        throw new BadRequestException('Invalid or expired verification code')
    }

    await userRepository.updateOne(
        { email },
        { $set: { isVerified: true } }
    )

    await redis.del(`email-verification:${email}`)

    return {
        email,
        isVerified: true
    }
}

export const login = async (data: LoginDTO) => {
    const email = data.email.toLowerCase()

    const user = await userRepository.findOne({ email })

    if (!user) {
        throw new UnauthorizedException('Invalid email or password')
    }

    const isPasswordCorrect = await comparePassword(
        data.password,
        user.password
    )

    if (!isPasswordCorrect) {
        throw new UnauthorizedException('Invalid email or password')
    }

    if (!user.isVerified) {
        throw new UnauthorizedException('Please verify your email first')
    }

    const accessToken = generateAccessToken(user._id.toString())

    return {
        accessToken,
        tokenType: 'Bearer',
        expiresIn: 3600,
        user: {
            id: user._id.toString(),
            firstName: user.firstName,
            lastName: user.lastName,
            email: user.email
        }
    }
}

    commentId: string,
    userId: string,
    data: UpdateCommentDTO
) => {

        {
            _id: new Types.ObjectId(commentId),
            userId: new Types.ObjectId(userId),
            isDeleted: false
        },
        {
            $set: {
                content: data.content
            }
        }
    )

    if (!comment) {
        throw new NotFoundException('Comment not found or you are not the owner')
    }

    return comment
}

export const deleteComment = async (
    commentId: string,
    userId: string
) => {
    const comment = await commentRepository.findOne({
        _id: new Types.ObjectId(commentId),
        isDeleted: false
    })

    if (!comment) {
        throw new NotFoundException('Comment not found')
    }

    const isCommentOwner = comment.userId.toString() === userId

    if (!isCommentOwner) {
        const post = await postRepository.findById(comment.postId.toString())

        if (!post || post.userId.toString() !== userId) {
            throw new UnauthorizedException(
                'Only the comment owner or post owner can delete this comment'
            )
        }
    }

    await commentRepository.findOneAndUpdate(
        {
            _id: comment._id,
            isDeleted: false
        },
        {
            $set: {
                isDeleted: true,
                deletedAt: new Date()
            }
        }
    )

    return {
        id: comment._id.toString(),
        isDeleted: true
    }
}
